<script>

const KEY="X7_DATA";

const defaults={
 dailyMoney:500,
 foodBudget:150,
 transportBudget:40,
 cigarettePrice:14,
 cigaretteLimit:3,
 debt:30000,
 originalDebt:30000,
 expenses:[],
 debtPayments:[],
 meals:[],
 sleepTarget:"00:30"
};

let data=load();

function load(){

 try{
  const saved=JSON.parse(localStorage.getItem(KEY));
  return {...defaults,...saved};
 }catch{
  return {...defaults};
 }

}

function save(){
 localStorage.setItem(KEY,JSON.stringify(data));
}

function today(){

 return new Date().toISOString().slice(0,10);

}

function money(n){

 return "₹"+Math.round(n).toLocaleString("en-IN");

}

function expensesToday(){

 return data.expenses.filter(x=>x.date===today());

}

function totalSpent(){

 return expensesToday()
 .reduce((a,b)=>a+b.amount,0);

}

function categorySpent(cat){

 return expensesToday()
 .filter(x=>x.category===cat)
 .reduce((a,b)=>a+b.amount,0);

}

function cigarettesToday(){

 return expensesToday()
 .filter(x=>x.category==="cigarette")
 .reduce((a,b)=>a+b.qty,0);

}

function update(){

 const spent=totalSpent();

 const remaining=Math.max(
  0,
  data.dailyMoney-spent
 );

 document.getElementById("balance").textContent=
  money(remaining);

 document.getElementById("received").textContent=
  money(data.dailyMoney);

 document.getElementById("spent").textContent=
  money(spent);

 document.getElementById("moneyProgress").style.width=
  Math.min(100,spent/data.dailyMoney*100)+"%";


 const foodLeft=Math.max(
  0,
  data.foodBudget-categorySpent("food")
 );

 const transportLeft=Math.max(
  0,
  data.transportBudget-categorySpent("transport")
 );

 document.getElementById("foodLeft").textContent=
  money(foodLeft)+" left";

 document.getElementById("transportLeft").textContent=
  money(transportLeft)+" left";


 const cig=cigarettesToday();

 document.getElementById("cigStatus").textContent=
  cig+" / "+data.cigaretteLimit;


 document.getElementById("debt").textContent=
  money(data.debt);

 const paid=
  data.originalDebt-data.debt;

 document.getElementById("debtPaid").textContent=
  money(paid);

 document.getElementById("debtProgress").style.width=
  Math.min(
   100,
   paid/data.originalDebt*100
  )+"%";


 const meals=
  data.meals.filter(x=>x.date===today()).length;

 document.getElementById("mealStatus").textContent=
  meals+" / 3";


 document.getElementById("sleepStatus").textContent=
  "Target "+formatTime(data.sleepTarget);


 renderActivity();

}

function formatTime(t){

 let [h,m]=t.split(":");
 h=Number(h);

 let suffix=h>=12?"PM":"AM";

 h=h%12||12;

 return `${h}:${m} ${suffix}`;

}


function expense(category){

 let label={
  food:"Food",
  transport:"Transport"
 }[category];

 let defaultAmount=
  category==="transport"?20:"";

 openForm(
  "Add "+label,
  category,
  defaultAmount
 );

}


function otherExpense(){

 openForm("Other spending","other","");

}


function openForm(title,category,amount){

 document.getElementById("modalTitle").textContent=
  title;

 document.getElementById("modalContent").innerHTML=`

  <div class="field">
   <label>AMOUNT</label>
   <input id="amount" type="number"
          value="${amount}"
          placeholder="₹ amount">
  </div>

  <div class="field">
   <label>NOTE</label>
   <input id="note" placeholder="What was it?">
  </div>

  <div class="actions">
   <button onclick="closeModal()">Cancel</button>
   <button class="primary"
           onclick="saveExpense('${category}')">
    Save
   </button>
  </div>
 `;

 document.getElementById("modal")
 .classList.remove("hidden");

}


function saveExpense(category){

 const amount=
  Number(document.getElementById("amount").value);

 if(!amount||amount<=0)return;

 const note=
  document.getElementById("note").value;

 data.expenses.push({

  date:today(),
  category,
  amount,
  qty:1,
  note:note||category,
  time:Date.now()

 });

 save();
 closeModal();
 update();

}


function cigarette(){

 const count=cigarettesToday();

 if(count>=data.cigaretteLimit){

  if(!confirm(
   "You've reached today's cigarette target.\n\n"+
   "Log another cigarette for "+money(data.cigarettePrice)+"?"
  ))return;

 }

 data.expenses.push({

  date:today(),
  category:"cigarette",
  amount:data.cigarettePrice,
  qty:1,
  note:"Cigarette",
  time:Date.now()

 });

 save();
 update();

}

function saveDebt(){

 let amount=
  Number(document.getElementById("debtAmount").value);

 if(!amount||amount<=0)return;

 amount=Math.min(amount,data.debt);

 data.debt-=amount;

  amount,
  date:today(),
  time:Date.now()

 });

 closeModal();
 update();

}


function checkFood(){

 const name=
  document.getElementById("foodName").value||
  "This";

 const price=
  Number(document.getElementById("foodPrice").value);

 if(!price)return;

 const remaining=
  Math.max(
   0,
   data.foodBudget-categorySpent("food")
  );

 let text="";

 if(price<=remaining){

  text=
   `🟢 <b>${name}</b> is within your food budget.<br>`+
   `You'll have ${money(remaining-price)} left.`;

 }else{

  text=
   `🔴 <b>${name}</b> is ₹${price-remaining} over your food budget.`;

 }

 document.getElementById("foodResult").innerHTML=
  `<div class="result">${text}</div>`;

}

function renderActivity(){

 const box=
  document.getElementById("activity");

 const items=
  expensesToday()
  .slice()
  .reverse();

 if(!items.length){

  box.innerHTML=
   `<div class="empty">Nothing logged yet. Keep control.</div>`;

  return;

 }

 box.innerHTML=
  items.map(x=>{

   let icon=
    x.category==="food"?"🍛":
    x.category==="transport"?"🚇":
    x.category==="cigarette"?"🚬":"🛍️";

   return `

    <div class="row">

     <div class="row-left">

      <div class="row-icon">${icon}</div>

      <div>
       <div class="row-name">
        ${x.note}
       </div>

       <div class="row-sub">
        ${new Date(x.time)
         .toLocaleTimeString([],
          {hour:"2-digit",minute:"2-digit"})}
       </div>

      </div>

     </div>

     <div class="amount negative">
      -${money(x.amount)}
     </div>

    </div>
   `;

  }).join("");

}
