    /* =======================================================
       CALCULATIONS
    ======================================================== */

    function calculate() {

      let income = 0;
      let expenses = 0;


      transactions.forEach(
        transaction => {

          if (
            transaction.type === "income"
          ) {

            income += transaction.amount;

          } else {

            expenses += transaction.amount;

          }

        }
      );


      const balance =
        income - expenses;


      document.getElementById(
        "income"
      ).textContent = money(income);


      document.getElementById(
        "expenses"
      ).textContent = money(expenses);


      document.getElementById(
        "balance"
      ).textContent = money(balance);

    }

        "Other"
      ];


      container.innerHTML = "";


      categories.forEach(
        category => {

          let total = 0;


          transactions.forEach(
            transaction => {

              if (
                transaction.category === category
              ) {

                total += transaction.amount;

              }

            }
          );


          const data =
            categoryData[category];


          const card =
            document.createElement("div");


          card.className =
            "category-card";


          card.innerHTML = `

            <div class="category-icon">
              ${data.icon}
            </div>

            <div class="category-name">
              ${category}
            </div>

            <div class="category-amount">
              ₹${money(total)}
            </div>

          `;


          container.appendChild(card);

        }
      );

    }


    /* =======================================================
       TRANSACTIONS
    ======================================================== */

    function renderTransactions() {

      const container =
        document.getElementById(
          "transactions"
        );


      container.innerHTML = "";


      if (
        transactions.length === 0
      ) {

        container.innerHTML = `

          <div class="empty">

            No transactions yet.

            <br>

            Start tracking your money with X7.

          </div>

        `;

      } else {

        transactions
          .slice()
          .reverse()
          .forEach(
            transaction => {

              const item =
                document.createElement(
                  "div"
                );


              item.className =
                "transaction";


              const data =
                categoryData[
                  transaction.category
                ] ||
                categoryData.Other;


              const sign =
                transaction.type === "income"
                  ? "+"
                  : "-";


              const amountClass =
                transaction.type === "income"
                  ? "income-text"
                  : "expense-text";


              item.innerHTML = `

                <div class="transaction-left">

                  <div class="transaction-icon">
                    ${data.icon}
                  </div>

                  <div class="transaction-info">

                    <div class="transaction-description">
                      ${escapeHTML(
                        transaction.description
                      )}
                    </div>

                    <div class="transaction-meta">

                      ${transaction.category}
                      •
                      ${transaction.date}

                    </div>

                  </div>

                </div>


                <div class="transaction-right">

                  <div
                    class="
                      transaction-amount
                      ${amountClass}
                    ">

                    ${sign} ₹${money(
                      transaction.amount
                    )}

                  </div>


                  <button
                    class="delete-button"
                    onclick="deleteTransaction(
                      '${transaction.id}'
                    )">

                    ×

                  </button>

                </div>

              `;


              container.appendChild(item);

            }
          );

      }


      const count =
        transactions.length;


      document.getElementById(
        "transactionCount"
      ).textContent =
        `${count} transaction${
          count === 1 ? "" : "s"
        }`;

    }


    /* =======================================================
       HTML ESCAPE
    ======================================================== */

    function escapeHTML(value) {

      const div =
        document.createElement(
          "div"
        );

      div.textContent = value;

      return div.innerHTML;

    }


    /* =======================================================
       TYPE
    ======================================================== */

    function setType(type) {

      currentType = type;


      document
        .getElementById("expenseType")
        .classList
        .toggle(
          "active",
          type === "expense"
        );


      document
        .getElementById("incomeType")
        .classList
        .toggle(
          "active",
          type === "income"
        );


      const categoryGroup =
        document.getElementById(
          "categoryGroup"
        );


      if (type === "income") {

        categoryGroup.style.display =
          "none";

      } else {

        categoryGroup.style.display =
          "block";

      }

    }


    /* =======================================================
       ADD TRANSACTION
    ======================================================== */

    function addTransaction() {

      const description =
        document.getElementById(
          "description"
        ).value.trim();


      const amount =
        parseFloat(
          document.getElementById(
            "amount"
          ).value
        );


      const category =
        document.getElementById(
          "category"
        ).value;


      if (!description) {

        alert(
          "Enter a description."
        );

        return;

      }


      if (
        !amount ||
        amount <= 0
      ) {

        alert(
          "Enter a valid amount."
        );

        return;

      }


      const transaction = {

        id:
          Date.now().toString(),

        description,

        amount,

        type:
          currentType,

        category:
          currentType === "income"
            ? "Income"
            : category,

        date:
          new Date().toLocaleDateString(
            "en-IN",
            {
              day: "numeric",
              month: "short",
              year: "numeric"
            }
          )

      };


      transactions.push(
        transaction
      );


      saveData();

      calculate();

      renderCategories();

      renderTransactions();

      closeAddModal();


      document.getElementById(
        "description"
      ).value = "";


      document.getElementById(
        "amount"
      ).value = "";


      setType("expense");

    }


    /* =======================================================
       DELETE
    ======================================================== */

    function deleteTransaction(id) {

      const confirmed =
        confirm(
          "Delete this transaction?"
        );


      if (!confirmed) {
        return;
      }


      transactions =
        transactions.filter(
          transaction =>
            transaction.id !== id
        );


      saveData();

      calculate();

      renderCategories();

      renderTransactions();

    }


    /* =======================================================
       RESET MONEY
    ======================================================== */

    function resetMoney() {

      transactions = [];


      localStorage.removeItem(
        "x7_transactions"
      );


      calculate();

      renderCategories();

      renderTransactions();

      closeResetModal();

    }
